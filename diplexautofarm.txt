local function tween(object, duration, properties)
    local animation = TweenService:Create(
        object,
        TweenInfo.new(
            duration,
            Enum.EasingStyle.Quint,
            Enum.EasingDirection.Out
        ),
        properties
    )

    animation:Play()
    return animation
end

--// Character handling
local function updateCharacter(newCharacter)
    cancelActivation()
    restoreAnchors()
    character = newCharacter
    humanoidRootPart = nil

    task.spawn(function()
        local root = newCharacter:WaitForChild(
            "HumanoidRootPart",
            10
        )

        if not root or interfaceClosed or character ~= newCharacter then
            return
        end

        humanoidRootPart = root

        if isAnchored then
            anchorOriginal[root] = root.Anchored
            root.Anchored = true
        end
    end)
end

    if not sliding then return end
    if (slideInput.UserInputType == Enum.UserInputType.Touch and input == slideInput)

            and input.UserInputType == Enum.UserInputType.MouseMovement) then

    if input == slideInput then
        sliding = false
        slideInput = nil
    end
end)

--// Purchase prompt detection
local function cleanPromptText(value)
    return trim(tostring(value or ""):gsub("<[^>]*>", "")
        :gsub("&nbsp;", " "):gsub("&amp;", "&"):gsub("%s+", " "))
end

local function purchaseVerb(text)
    local lower = string.lower(cleanPromptText(text))
    return lower:match("^purchase%f[%W]") ~= nil
        or lower:match("^buy%f[%W]") ~= nil
end

local function isPurchasePrompt(prompt)
    if not prompt or not prompt:IsA("ProximityPrompt") then return false end
    -- ActionText is authoritative when it names a different operation.
    local action = cleanPromptText(prompt.ActionText)
    if purchaseVerb(action) then return true end
    if action ~= "" then return false end
    return purchaseVerb(prompt.Name) or purchaseVerb(prompt.ObjectText)
end

local function getPurchaseName(prompt)
    local object = cleanPromptText(prompt.ObjectText)
    if object ~= "" and not purchaseVerb(object) then return object end
    for _, value in ipairs({prompt.ActionText, prompt.ObjectText, prompt.Name}) do
        local text = cleanPromptText(value)
        if purchaseVerb(text) then
            local name = trim(text:gsub("^%a+", ""):gsub("^%s*[:%-]%s*", ""))
            name = name:gsub("^%[", ""):gsub("%]$", "")
            if name ~= "" then return name end
        end
    end
    local model = prompt:FindFirstAncestorWhichIsA("Model")
    return model and model.Name or "Brainrot"
end

--// Prompt position and range helpers
local function getPromptWorldPosition(prompt)
    if not prompt or not prompt.Parent then
        return nil
    end

    local holder = prompt.Parent

    if holder:IsA("Attachment") then
        return holder.WorldPosition
    end

    if holder:IsA("BasePart") then
        return holder.Position
    end

    if holder:IsA("Model") then
        return holder:GetPivot().Position
    end

    local attachment = holder:FindFirstAncestorWhichIsA("Attachment")

    if attachment then
        return attachment.WorldPosition
    end

    local part = holder:FindFirstAncestorWhichIsA("BasePart")

    if part then
        return part.Position
    end

    local model = holder:FindFirstAncestorWhichIsA("Model")

    if model then
        return model:GetPivot().Position
    end

    return nil
end

local function getRootPart()
    local root = humanoidRootPart

    if root and root.Parent then
        return root
    end

    if character then
        root = character:FindFirstChild("HumanoidRootPart")
    end

    humanoidRootPart = root
    return root
end

local function getPromptDistance(prompt)
    local root = getRootPart()
    local promptPosition = getPromptWorldPosition(prompt)

    if not root or not promptPosition then
        return math.huge
    end

    return (root.Position - promptPosition).Magnitude
end

local function isPromptInRange(prompt)
    return getPromptDistance(prompt)
        <= (AUTO_BUY_RANGE + DISTANCE_TOLERANCE)
end

local function improvePromptRange(prompt)
    if not prompt or not prompt.Parent then
        return
    end

    if not originalPromptSettings[prompt] then
        originalPromptSettings[prompt] = {
            MaxActivationDistance = prompt.MaxActivationDistance,
            RequiresLineOfSight = prompt.RequiresLineOfSight
        }
    end

    if prompt.MaxActivationDistance < AUTO_BUY_RANGE then
        prompt.MaxActivationDistance = AUTO_BUY_RANGE
    end

    if prompt.RequiresLineOfSight then
        prompt.RequiresLineOfSight = false
    end
end

local function restorePrompt(prompt)
    local original = originalPromptSettings[prompt]

    if not original or not prompt or not prompt.Parent then
        originalPromptSettings[prompt] = nil
        return
    end

    prompt.MaxActivationDistance = original.MaxActivationDistance
    prompt.RequiresLineOfSight = original.RequiresLineOfSight
    originalPromptSettings[prompt] = nil
end

local function registerPrompt(instance)
    if not instance or not instance:IsA("ProximityPrompt") then
        return
    end

    -- Track every prompt because some games fill ActionText/ObjectText later.
    trackedPrompts[instance] = true

    if autoBuyEnabled and isPurchasePrompt(instance) then
        improvePromptRange(instance)
    end
end

local function registerExistingPrompts()
    for _, instance in ipairs(workspace:GetDescendants()) do
        registerPrompt(instance)
    end
end

restoreAllPrompts = function()
    for prompt in pairs(originalPromptSettings) do
        restorePrompt(prompt)
    end
end

--// Serialized activation: never overlap holds or claim a confirmed purchase.
local function activatePurchasePrompt(prompt)
    if activePrompt or interfaceClosed or not autoBuyEnabled then return end
    if (promptCooldowns[prompt] or 0) > os.clock() then return end
    if not prompt:IsDescendantOf(workspace) or not prompt.Enabled
        or not isPurchasePrompt(prompt) or not isPromptInRange(prompt) then return end
    activePrompt = prompt
    local token = generation
    local root = getRootPart()
    task.spawn(function()
        local success, failure = pcall(function()
            improvePromptRange(prompt)
            setStatus("Trying " .. getPurchaseName(prompt), COLORS.PurpleLight)
            prompt:InputHoldBegin()
            local started = os.clock()
            repeat
                task.wait(0.03)
                if interfaceClosed or not autoBuyEnabled or token ~= generation
                    or root ~= getRootPart() or not prompt:IsDescendantOf(workspace)
                    or not prompt.Enabled or not isPurchasePrompt(prompt)
                    or not isPromptInRange(prompt) then return end
            until os.clock() - started >= math.max(prompt.HoldDuration, 0.03) + 0.04
        end)
        pcall(function() prompt:InputHoldEnd() end)
        if activePrompt == prompt then activePrompt = nil end
        if prompt:IsDescendantOf(workspace) then
            promptCooldowns[prompt] = os.clock() + (success and PROMPT_COOLDOWN or 2)
        end
        if not success and not interfaceClosed then
            setStatus("Activation failed - retrying", COLORS.Red)
            warn("[Trait Farm] " .. tostring(failure))
        end
    end)
end

connect(ProximityPromptService.PromptShown, function(prompt)
    if not interfaceClosed then registerPrompt(prompt) end
end)
connect(workspace.DescendantAdded, function(instance)
    if instance:IsA("ProximityPrompt") then registerPrompt(instance) end
end)
connect(workspace.DescendantRemoving, function(instance)
    if not instance:IsA("ProximityPrompt") then return end
    if activePrompt == instance then cancelActivation() end
    restorePrompt(instance)
    trackedPrompts[instance] = nil
    promptCooldowns[instance] = nil
end)

-- Listeners attach before the initial scan so newly streamed prompts are caught.
registerExistingPrompts()
local scanAccumulator = 0
connect(RunService.Heartbeat, function(deltaTime)
    if interfaceClosed or not autoBuyEnabled then return end
    scanAccumulator = scanAccumulator + deltaTime
    if scanAccumulator < SCAN_INTERVAL then return end
    scanAccumulator = 0
    local nearestPrompt, nearestDistance = nil, math.huge
    local detected, nearby = 0, 0
    local now = os.clock()
    for prompt in pairs(trackedPrompts) do
        if not prompt:IsDescendantOf(workspace) then
            restorePrompt(prompt)
            trackedPrompts[prompt] = nil
            promptCooldowns[prompt] = nil
        elseif isPurchasePrompt(prompt) then
            detected = detected + 1
            if prompt.Enabled then
                local distance = getPromptDistance(prompt)
                if distance <= AUTO_BUY_RANGE + DISTANCE_TOLERANCE then
                    nearby = nearby + 1
                    if distance < nearestDistance and (promptCooldowns[prompt] or 0) <= now then
                        nearestPrompt, nearestDistance = prompt, distance
                    end
                end
            end
        else
            restorePrompt(prompt)
        end
        -- Disabled prompts stay tracked and are re-evaluated when enabled.
    end
    if not activePrompt then
        setStatus(string.format("%d detected / %d nearby", detected, nearby), COLORS.Green)
        if nearestPrompt then activatePurchasePrompt(nearestPrompt) end
    end
end)

-- Fit the same interface on phones and respond to viewport changes.
local mainScale = Instance.new("UIScale", mainFrame)
local shadowScale = Instance.new("UIScale", shadow)
local viewportConnection
refreshScale = function()
    local camera = workspace.CurrentCamera
    if not camera then return end
    local viewport = camera.ViewportSize
    local fittingScale = math.min(
        math.max(0.1, (viewport.X - 24) / 384),
        math.max(0.1, (viewport.Y - 24) / 354))
    local scale = math.min(requestedScale, fittingScale)
    mainScale.Scale, shadowScale.Scale = scale, scale
end
local function bindViewport()
    if viewportConnection then viewportConnection:Disconnect() end
    local camera = workspace.CurrentCamera
    refreshScale()
    if camera then viewportConnection = camera:GetPropertyChangedSignal("ViewportSize"):Connect(refreshScale) end
end
bindViewport()
connect(workspace:GetPropertyChangedSignal("CurrentCamera"), bindViewport)

--// Smooth dragging
local dragging = false
local dragInput = nil
local dragStart = nil
local frameStart = nil
local targetPosition = mainFrame.Position

connect(header.InputBegan, function(input)
    local inputType = input.UserInputType

    if inputType ~= Enum.UserInputType.MouseButton1
        and inputType ~= Enum.UserInputType.Touch then
        return
    end

    dragging = true
    dragStart = input.Position
    frameStart = mainFrame.Position


end)

connect(UserInputService.InputEnded, function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input == dragInput then
        dragging = false
    end
end)

connect(header.InputChanged, function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

connect(UserInputService.InputChanged, function(input)
    if not dragging or input ~= dragInput then
        return
    end

    local delta = input.Position - dragStart

    targetPosition = UDim2.new(
        frameStart.X.Scale,
        frameStart.X.Offset + delta.X / mainScale.Scale,
        frameStart.Y.Scale,
        frameStart.Y.Offset + delta.Y / mainScale.Scale
    )
end)

connect(RunService.RenderStepped, function(deltaTime)
    if interfaceClosed then
        return
    end

    local smoothness = math.clamp(
        deltaTime * 22,
        0,
        1
    )

    mainFrame.Position = mainFrame.Position:Lerp(
        targetPosition,
        smoothness
    )

    shadow.Position = mainFrame.Position + UDim2.fromOffset(6, 8)
end)
